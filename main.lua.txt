local function isNetworkAvailable()
    local cm = this.getSystemService(Context.CONNECTIVITY_SERVICE)
    if cm then
        if Build.VERSION.SDK_INT >= Build.VERSION_CODES.M then
            local network = cm.getActiveNetwork()
            if network then
                local caps = cm.getNetworkCapabilities(network)
                if caps then
                    return caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) and
                           caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED)
                end
            end
        else
            local ni = cm.getActiveNetworkInfo()
            return ni and ni.isConnected()
        end
    end
    return false
end

-- ============================================
-- HTTP HELPERS
-- ============================================
local function httpGet(url, timeoutMs)
    timeoutMs = timeoutMs or 5000
    local result = nil
    local success = pcall(function()
        local conn = URL(url).openConnection()
        conn.setConnectTimeout(timeoutMs)
        conn.setReadTimeout(timeoutMs)
        conn.setRequestMethod("GET")
        conn.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android 12) AppleWebKit/537.36")
        conn.setRequestProperty("Accept", "application/json, text/plain, */*")
        local responseCode = conn.getResponseCode()
        if responseCode == 200 then
            local reader = BufferedReader(InputStreamReader(conn.getInputStream(), "UTF-8"))
            local sb = StringBuilder()
            local line = reader.readLine()
            while line do sb.append(line).append("\n"); line = reader.readLine() end
            reader.close()
            result = sb.toString()
        end
        conn.disconnect()
    end)
    if success then return result end
    return nil
end

local function httpPost(url, postData, timeoutMs)
    timeoutMs = timeoutMs or 5000
    local result = nil
    local success = pcall(function()
        local conn = URL(url).openConnection()
        conn.setConnectTimeout(timeoutMs)
        conn.setReadTimeout(timeoutMs)
        conn.setRequestMethod("POST")
        conn.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android 12) AppleWebKit/537.36")
        conn.setRequestProperty("Content-Type", "application/json")
        conn.setDoOutput(true)
        local os = conn.getOutputStream()
        local bytes = postData:getBytes("UTF-8")
        os.write(bytes, 0, #bytes)
        os.flush()
        os.close()
        local responseCode = conn.getResponseCode()
        if responseCode == 200 then
            local reader = BufferedReader(InputStreamReader(conn.getInputStream(), "UTF-8"))
            local sb = StringBuilder()
            local line = reader.readLine()
            while line do sb.append(line).append("\n"); line = reader.readLine() end
            reader.close()
            result = sb.toString()
        end
        conn.disconnect()
    end)
    if success then return result end
    return nil
end

local function httpGetWithStatus(url, timeoutMs)
    timeoutMs = timeoutMs or 5000
    local result = {code = -1, body = nil, error = nil}
    local success = pcall(function()
        local conn = URL(url).openConnection()
        conn.setConnectTimeout(timeoutMs)
        conn.setReadTimeout(timeoutMs)
        conn.setRequestMethod("GET")
        conn.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android 12) AppleWebKit/537.36")
        result.code = conn.getResponseCode()
        if result.code == 200 then
            local reader = BufferedReader(InputStreamReader(conn.getInputStream(), "UTF-8"))
            local sb = StringBuilder()
            local line = reader.readLine()
            while line do sb.append(line).append("\n"); line = reader.readLine() end
            reader.close()
            result.body = sb.toString()
        end
        conn.disconnect()
    end)
    if not success then result.error = "Connection failed" end
    return result
end

local function urlEncode(text)
    local ok, enc = pcall(function() return URLEncoder.encode(text, "UTF-8") end)
    if ok then return enc end
    return text
end

-- ============================================
-- UTF8 Safe Decode
-- ============================================
local function decodeUnicode(str)
    if not str then return str end
    local ok, result = pcall(function()
        return str:gsub("\\u(%x%x%x%x)", function(h) 
            local n = tonumber(h, 16)
            if n and n > 0 then
                local ok2, ch = pcall(string.char, n)
                if ok2 then return ch end
            end
            return ""
        end)
    end)
    if ok then return result end
    return str
end

-- ✅ AI ENGINE 1: GOOGLE (Super Powerful)
-- ============================================
local function engineGoogle(text, src, tgt)
    local url = "https://translate.googleapis.com/translate_a/single?client=gtx&sl=" .. src .. "&tl=" .. tgt .. "&dt=t&q=" .. urlEncode(text)
    local response = httpGet(url, 5000)
    if not response then return nil end
    
    local translated = ""
    local segmentCount = 0
    
    for trans, orig in response:gmatch('%["(.-)","(.-)"') do
        if trans and trans ~= "" then
            if not trans:match("^%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x%x$") then
                translated = translated .. trans
                segmentCount = segmentCount + 1
            end
        end
    end
    
    if translated ~= "" and segmentCount > 0 then
        translated = translated:gsub('\\"', '"'):gsub("\\n", " ")
        translated = decodeUnicode(translated)
        translated = translated:gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "")
        return translated
    end
    return nil
end

-- ✅ AI ENGINE 2: MYMEMORY (Super Powerful)
-- ============================================
local function engineMyMemory(text, src, tgt)
    local url = "https://api.mymemory.translated.net/get?q=" .. urlEncode(text) .. "&langpair=" .. src .. "|" .. tgt
    local response = httpGet(url, 5000)
    if not response then return nil end
    
    local translated = response:match('"translatedText":"(.-)"')
    if not translated or translated == "" then return nil end
    
    if translated:find("MYMEMORY WARNING") 
        or translated:find("QUERY LENGTH LIMIT")
        or translated:find("INVALID")
        or #translated < 2 then
        return nil
    end
    
    translated = decodeUnicode(translated)
    return translated
end

local function engineMicrosoft(text, src, tgt)

    local postData = '{"Text":"' .. text:gsub('"', '\\"'):gsub("\n", " ") .. '"}'

    local translated = response:match('"translations":%[{"text":"(.-)"')
    if not translated or translated == "" then
        translated = response:match('"text":"(.-)"')

    if translated and translated ~= "" then
        translated = decodeUnicode(translated)
        return translated
    end
    return nil
end

-- ============================================
-- ✅ SUPER POWERFUL SCORING (12 Levels)
-- ============================================
local function scoreTranslation(text, src, tgt, originalText)
    if not text or text == "" then return -10000 end
    
    local score = 0
    local len = #text
    local origLen = #(originalText or "")
    
    if len < 2 then score = score - 1000 end
    if len > 3000 then score = score - 200 end
    
    if src == "ur" or src == "ar" then
        local urduCount = 0
        for _ in text:gmatch("[\216-\219]") do urduCount = urduCount + 1 end
        if urduCount > 0 then score = score - (urduCount * 50) end
    elseif src == "hi" then
        local hindiCount = 0
        for _ in text:gmatch("[\224-\239]") do hindiCount = hindiCount + 1 end
        if hindiCount > 0 then score = score - (hindiCount * 50) end
    end
    
    if tgt == "en" then
        if text:match("%a") then score = score + 200 end
        
        local commonWords = {
            "the", "a", "an", "is", "are", "was", "were", "be", "been", "being",
            "have", "has", "had", "do", "does", "did", "will", "would", "shall",
            "should", "can", "could", "may", "might", "must", "and", "or", "but",
            "to", "of", "in", "on", "at", "for", "with", "by", "from", "into",
            "about", "as", "up", "down", "out", "off", "over", "under", "again",
            "you", "I", "we", "they", "he", "she", "it", "this", "that", "these",
            "those", "what", "which", "who", "whom", "whose", "where", "when",
            "why", "how", "not", "no", "yes", "very", "so", "too", "just", "only"
        }
        local lower = text:lower()
        local wordCount = 0
        for _, w in ipairs(commonWords) do
            local _, count = lower:gsub("%f[%a]" .. w .. "%f[%A]", "")
            wordCount = wordCount + count
        end
        score = score + (wordCount * 25)
        
        local totalWords = 0
        for _ in text:gmatch("%a+") do totalWords = totalWords + 1 end
        if totalWords < 2 then score = score - 500 end
        
        if text:match("^%u") then score = score + 30 end
        if text:match("[%.%?!]%s*$") then score = score + 50 end
        
    elseif tgt == "ur" or tgt == "ar" then
        if text:find("[\216-\219]") then score = score + 200 end
    end
    
    if text:find("WARNING") or text:find("LIMIT") or text:find("ERROR") or text:find("INVALID") then 
        score = score - 5000 
    end
    
    if origLen > 0 and len > 0 then
        local ratio = len / origLen
        if ratio >= 0.4 and ratio <= 2.5 then 
            score = score + 200 
        elseif ratio < 0.3 or ratio > 3.0 then
            score = score - 300
        end
    end
    
    if text:match("(%S+)%s+%1%s+%1") then score = score - 500 end
    
    if tgt == "en" then
        local firstChar = text:sub(1,1)
        if firstChar:match("%a") and firstChar == firstChar:lower() then
            score = score - 30
        end
    end
    
    return score
end

-- ✅ MULTI-AI TRANSLATE (3 Engines + Default Mode)
-- ============================================
local function multiAITranslate(text, src, tgt, callback)
    if not text or text == "" then callback("", "empty"); return end
    if src == tgt then callback(text, "same_lang"); return end
    if not isNetworkAvailable() then callback(nil, "no_network"); return end
    
    local resGoogle = nil
    local resMyMemory = nil
    local resMicrosoft = nil
    
    -- ✅ DEFAULT MODE — Teenon engines ek saath
    if config.useDefault then
        if config.useGoogle then
            local ok1, r1 = pcall(engineGoogle, text, src, tgt)
            if ok1 and r1 and r1 ~= "" then resGoogle = r1 end
        end
        
        if config.useMyMemory then
            local ok2, r2 = pcall(engineMyMemory, text, src, tgt)
            if ok2 and r2 and r2 ~= "" then resMyMemory = r2 end
        end
        
        if config.useMicrosoft then
            local ok3, r3 = pcall(engineMicrosoft, text, src, tgt)
            if ok3 and r3 and r3 ~= "" then resMicrosoft = r3 end
        end
    else
        -- User ki marzi
        if config.useGoogle then
            local ok1, r1 = pcall(engineGoogle, text, src, tgt)
            if ok1 and r1 and r1 ~= "" then resGoogle = r1 end
        end
        
        if config.useMyMemory then
            local ok2, r2 = pcall(engineMyMemory, text, src, tgt)
            if ok2 and r2 and r2 ~= "" then resMyMemory = r2 end
        end
        
        if config.useMicrosoft then
            local ok3, r3 = pcall(engineMicrosoft, text, src, tgt)
            if ok3 and r3 and r3 ~= "" then resMicrosoft = r3 end
        end
    end
    
    -- Best result choose karo
    local candidates = {}
    if resGoogle then table.insert(candidates, {text = resGoogle, engine = "Google"}) end
    if resMyMemory then table.insert(candidates, {text = resMyMemory, engine = "MyMemory"}) end
    if resMicrosoft then table.insert(candidates, {text = resMicrosoft, engine = "Microsoft"}) end
    
    if #candidates == 0 then
        callback(nil, "all_failed"); return
    end
    
    if #candidates == 1 then
        callback(candidates[1].text, candidates[1].engine); return
    end
    
    -- Multiple — score compare
    local best = candidates[1]
    local bestScore = scoreTranslation(best.text, src, tgt, text)
    
    for i = 2, #candidates do
        local sc = scoreTranslation(candidates[i].text, src, tgt, text)
        if sc > bestScore + 200 then
            best = candidates[i]
            bestScore = sc
        end
    end
    
    callback(best.text, best.engine .. " (best)")
end

-- ============================================
-- CHUNK TRANSLATION
-- ============================================
local function chunkedTranslate(text, src, tgt, callback)
    if not text or text == "" then callback("", "empty"); return end
    if #text <= 400 then multiAITranslate(text, src, tgt, callback); return end
    
    local chunks = {}
    local sentenceEndings = {".", "!", "?", "۔", "؟", "،", "。", "？", "\n"}
    local remaining = text
    
    while #remaining > 0 do
        local bestPos = nil
        for _, endChar in ipairs(sentenceEndings) do
            local pos = remaining:find(endChar, 1, true)
            if pos and pos <= 400 and (not bestPos or pos < bestPos) then
                bestPos = pos
            end
        end
        
        if not bestPos then
            if #remaining <= 400 then
                table.insert(chunks, remaining); remaining = ""
            else
                local part = remaining:sub(1, 400)
                local lastSpace = part:match(".*()%s")
                if lastSpace and lastSpace > 200 then part = remaining:sub(1, lastSpace - 1) end
                table.insert(chunks, part)
                remaining = remaining:sub(#part + 1):gsub("^%s+", "")
            end
        else
            local part = remaining:sub(1, bestPos)
            table.insert(chunks, part)
            remaining = remaining:sub(bestPos + 1):gsub("^%s+", "")
        end
    end
    
    local results = {}
    local usedEngine = nil
    
    for _, chunk in ipairs(chunks) do
        local chunkResult = nil
        local chunkEngine = nil
        multiAITranslate(chunk, src, tgt, function(translated, engine)
            if translated and translated ~= "" then
                chunkResult = translated
                chunkEngine = engine
            end
        end)
        if chunkResult then
            table.insert(results, chunkResult)
            if not usedEngine then usedEngine = chunkEngine end
        else
            table.insert(results, chunk)
        end
    end
    
    if #results > 0 then
        callback(table.concat(results, " "), usedEngine or "Chunked")
    else
        callback(nil, "chunked_failed")
    end
end

local function loadDictionary()
    for lang, _ in pairs(wordBanks) do
        local saved = pref.getString("dict_" .. lang, "")
        if saved and saved ~= "" then
            for pair in saved:gmatch("[^|]+") do
                local word, meaning = pair:match("^(.-)=(.+)$")
                if word and meaning then
                    wordBanks[lang][word] = meaning
                end
            end
        end
    end
end

local function saveDictionary(lang)
    if not wordBanks[lang] then return end
    local parts = {}
    for word, meaning in pairs(wordBanks[lang]) do
        table.insert(parts, word .. "=" .. meaning)
    end
    pref.edit().putString("dict_" .. lang, table.concat(parts, "|")).apply()
end

local function getWordCount(lang)
    if not wordBanks[lang] then return 0 end
    local count = 0
    for _ in pairs(wordBanks[lang]) do count = count + 1 end
    return count
end

-- ✅ AUTO-LEARNING (Phrase + Word Level)
local function autoLearnWords(originalText, translatedText, srcLang, tgtLang)
    if not originalText or not translatedText then return end
    
    local srcClean = originalText:gsub("^%s+", ""):gsub("%s+$", "")
    local tgtClean = translatedText:gsub("^%s+", ""):gsub("%s+$", "")
    
    if #srcClean > 3 and #srcClean < 60 and #tgtClean > 3 and #tgtClean < 100 then
        if wordBanks[srcLang] and not wordBanks[srcLang][srcClean] then
            wordBanks[srcLang][srcClean] = tgtClean
            saveDictionary(srcLang)
        end
        if wordBanks[tgtLang] and not wordBanks[tgtLang][tgtClean] then
            wordBanks[tgtLang][tgtClean] = srcClean
            saveDictionary(tgtLang)
        end
    end
    
    local srcWords = {}
    for w in originalText:gmatch("[^%s%p]+") do
        table.insert(srcWords, w)
    end
    
    local tgtWords = {}
    for w in translatedText:gmatch("[^%s%p]+") do
        table.insert(tgtWords, w)
    end
    
    if #srcWords == #tgtWords and #srcWords > 0 then
        for i = 1, #srcWords do
            local srcWord = srcWords[i]
            local tgtWord = tgtWords[i]
            if #srcWord > 3 and #tgtWord > 2 then
                if wordBanks[srcLang] and not wordBanks[srcLang][srcWord] then
                    wordBanks[srcLang][srcWord] = tgtWord
                end
                if wordBanks[tgtLang] and not wordBanks[tgtLang][tgtWord] then
                    wordBanks[tgtLang][tgtWord] = srcWord
                end
            end
        end
        saveDictionary(srcLang)
        saveDictionary(tgtLang)
    end
end

local function applyWordChecker(text)
    if not text or text == "" then return text end
    if not config.wordChecker then return text end
    
    local result = text
    local words = {}
    for word in result:gmatch("%S+") do
        table.insert(words, word)
    end
    
    local fixed = {}
    for _, word in ipairs(words) do
        local lower = word:lower()
        local punct = ""
        local clean = lower:gsub("^%W+", ""):gsub("%W+$", "")
        if clean ~= lower then punct = lower:match("%W+$") or ""; lower = clean end
        
        -- Check if valid English word
        if validEnglishWords[lower] then
            table.insert(fixed, word)  -- Keep as is
        elseif wordCheckerDict[lower] then
            -- Fix only if wrong
            local rep = wordCheckerDict[lower]
            if word:sub(1,1):match("%u") then 
                rep = rep:sub(1,1):upper() .. rep:sub(2) 
            end
            table.insert(fixed, rep .. punct)
        else
            table.insert(fixed, word)  -- Keep as is
        end
    end
    
    return table.concat(fixed, " ")
end

local function deepPunctuationAnalysis(text, lang)
  if not text or text == "" then return text end
  if not config.autoPunc then return text end
  
  local rules = punctuationRules[lang] or punctuationRules["en-US"]
  local qRules = questionRules[lang] or questionRules["en-US"]
  local result = text
  
  if result:find("[.!?؟!۔،。？]%s*$") then return result end
  
  local low = result:lower()
  local firstWord = low:match("^(%S+)") or ""
  firstWord = firstWord:gsub("^[%p%s]+", ""):gsub("[%p%s]+$", "")
  local isQuestion = false
  
  for _, qw in ipairs(qRules.startWords or {}) do
    if firstWord == qw:lower() then isQuestion = true; break end
  end
  
  if not isQuestion then
    for _, qw in ipairs(qRules.strongWords or {}) do
      local qwLow = qw:lower()
      if low == qwLow or low:find("^" .. qwLow .. "%s") or low:find("%s" .. qwLow .. "%s") then
        isQuestion = true; break
      end
    end
  end
  
  if not isQuestion and low:find("[?؟？]") then isQuestion = true end
  
  if isQuestion then
    result = result .. rules.question_end
  else
    result = result .. rules.sentence_end
  end
  
  return result
end

local function cleanText(text)
  if not text then return "" end
  return text:gsub("[%[%]{}()<>]", "")
             :gsub("[%*%_%~]", "")
             :gsub("[ \t]+", " ")
             :gsub("^%s+", "")
             :gsub("%s+$", "")
end

local function applyAutoCorrect(text, lang)
  if not text then return "" end
  local result = text
  local tbl = nil
  if lang == "en-US" then tbl = englishAutoCorrect
  elseif lang == "ur-PK" then tbl = urduAutoCorrect
  elseif lang == "hi-IN" then tbl = hindiAutoCorrect
  elseif lang == "ar-SA" then tbl = arabicAutoCorrect
  end
  if tbl then
    for wrong, correct in pairs(tbl) do
      local pattern = "(%f[%a])" .. wrong:gsub("(%W)", "%%%1") .. "(%f[%W])"
      result = result:gsub(pattern, function(p, s) return p .. correct .. s end)
    end
  end
  return result
end

local function applyFinalTouches(text, lang)
  local result = text
  result = deepPunctuationAnalysis(result, lang)
  
  if config.useEmoji then
    local emoji = deepContextEmojiDetect(result)
    if emoji ~= "" then result = result .. " " .. emoji end
  end
  
  result = cleanText(result)
  if lang == "en-US" and #result > 0 then
    local fc = result:sub(1,1)
    if fc:match("%a") then result = fc:upper() .. result:sub(2) end
  end
  return result
end

-- ============================================
-- 🧠 MAIN AI PROCESSING (Translation-Focused)
-- ============================================
local function processTextWithAILevel(text, lang, callback)
  if not text or text == "" then callback(""); return end
  
  -- Step 1: AI Verify (spelling, context, duplicates, word checker)
  text = aiVerifyText(text, lang)
  
  -- Step 2: Clean
  local result = cleanText(text)
  
  -- Step 3: Auto-correct
  result = applyAutoCorrect(result, lang)
  
  -- Step 4: Grammar Fix — Sirf agar user ne enable kiya
  if lang == "en-US" and config.grammarFix and config.aiRewriteEnabled then
    result = applyGrammarFix(result)
  end
  
  -- Step 5: English-specific fixes
  if lang == "en-US" then
    result = repairMergedWords(result)
    result = fixApostrophes(result)
    result = smartSpaceInsertion(result)
  end
  
  -- Step 6: AI Rewrite — Sirf agar user ne enable kiya
  if lang == "en-US" and config.englishLevel ~= "standard" and config.aiRewriteEnabled then
    local withPunc = applyFinalTouches(result, lang)
    aiRewriteEnglish(withPunc, config.englishLevel, function(rewritten)
      callback(rewritten or withPunc)
    end)
  else
    local final = applyFinalTouches(result, lang)
    callback(final)
  end
end-- ============================================

local pendingJob = nil

local function insertTextDirect(editTextRef, text)
    if not editTextRef then 
        speak("Text box not found")
        return false 
    end
    local clean = cleanText(text)
    local success = pcall(function()
        local currentText = ""
        pcall(function() currentText = editTextRef.getText().toString() end)
        if currentText and currentText ~= "" then
            service.insertText(editTextRef, " " .. clean)
        else
            service.insertText(editTextRef, clean)
        end
    end)
    return success
end

-- ============================================
-- 🎯 SCHEDULE PENDING TRANSLATION
-- ============================================
local function schedulePendingTranslation(text, src, tgt, mode, capturedEditText)
    if not text or text == "" then return end
    pendingJob = {
        text = text, src = src, tgt = tgt, mode = mode,
        editText = capturedEditText
    }
    
    mainHandler.postDelayed(function()
        local job = pendingJob
        if not job then return end
        pendingJob = nil
        if not job.editText then
            speak("Text box lost. Please tap on text box again.")
            return
        end
        
        if job.mode == "translate" then
            if not isNetworkAvailable() then
                speak("No internet. Please turn on WiFi or data.")
                return
            end
            
            speak("Translating")
            
            mainHandler.postDelayed(function()
                chunkedTranslate(job.text, job.src, job.tgt, function(translated, engine)
                    if translated and translated ~= "" then
                        pcall(function()
                            autoLearnWords(job.text, translated, job.src, job.tgt)
                        end)
                        
                        local finalText = translated
                        
                        processTextWithAILevel(finalText, job.tgt, function(processed)
                            mainHandler.postDelayed(function()
                                local ok = insertTextDirect(job.editText, processed)
                                if ok and config.readAfterType then 
                                    speak(processed)
                                elseif ok then
                                    speak("Text added")
                                end
                            end, 50)
                        end)
                    else
                        speak("Translation failed. Please try again.")
                    end
                end)
            end, 100)
        else
            processTextWithAILevel(job.text, job.src, function(processed)
                mainHandler.postDelayed(function()
                    local ok = insertTextDirect(job.editText, processed)
                    if ok and config.readAfterType then 
                        speak(processed) 
                    elseif ok then
                        speak("Text added")
                    end
                end, 50)
            end)
        end
    end, 300)
end

local currentRecognizer = nil
local isListening = false
local silenceTimer = nil
local lastPartialText = ""
local SILENCE_DURATION = 1500
local isTranslating = false

local function resetSilenceTimer(recognizer)
  if silenceTimer then mainHandler.removeCallbacks(silenceTimer); silenceTimer = nil end
  silenceTimer = Runnable{
    run = function()
      if isListening and recognizer and not isTranslating then 
          pcall(function() recognizer.stopListening() end) 
      end
      silenceTimer = nil
    end
  }
  mainHandler.postDelayed(silenceTimer, SILENCE_DURATION)
end

local function cleanupOldRecognizer()
    if silenceTimer then 
        mainHandler.removeCallbacks(silenceTimer)
        silenceTimer = nil 
    end
    if currentRecognizer then
        pcall(function() currentRecognizer.stopListening() end)
        pcall(function() currentRecognizer.cancel() end)
        pcall(function() currentRecognizer.destroy() end)
        currentRecognizer = nil
    end
    isListening = false
    lastPartialText = ""
    isTranslating = false
end

-- 🎤 START VOICE RECOGNITION
-- ============================================
function startVoice(lang)
    cleanupOldRecognizer()
    
    if not service then 
        speak("Service not available. Please turn on accessibility service.")
        return 
    end
    
    local editText = service.getEditText()
    if not editText then 
        speak("Please open a text box first. Tap on any text box.")
        return 
    end
    
    local isDynamicTranslate = (lang == "__DYNAMIC_TRANSLATE__")
    local actualLang = lang
    local srcLang, tgtLang = config.sourceLang, config.targetLang
    
    if isDynamicTranslate then
        if not isNetworkAvailable() then
            speak("No internet. Please turn on WiFi or data.")
            return
        end
        actualLang = getAndroidLocale(srcLang)
    end
    
    local srcName = getLangName(srcLang)
    local tgtName = getLangName(tgtLang)
    
    if isDynamicTranslate then
        speak(srcName .. " to " .. tgtName .. " translation mode.")
    elseif actualLang == "ur-PK" then
        speak("Urdu typing mode.")
    elseif actualLang == "hi-IN" then
        speak("Hindi typing mode.")
    elseif actualLang == "ar-SA" then
        speak("Arabic typing mode.")
    elseif actualLang == "pa-IN" then
        speak("Punjabi typing mode.")
    elseif actualLang == "en-US" then
        speak("English typing mode.")
    else
        speak("Listening")
    end
    
    local recognizer = SpeechRecognizer.createSpeechRecognizer(this)
    if not recognizer then 
        speak("Speech not available")
        return 
    end
    
    currentRecognizer = recognizer
    isListening = true
    lastPartialText = ""
    pendingJob = nil
    isTranslating = false
    
    local intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH)
    intent.putExtra(RecognizerIntent.EXTRA_LANGUAGE, actualLang)
    intent.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
    intent.putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3)
    intent.putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
    intent.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_MINIMUM_LENGTH_MILLIS, 500)
    intent.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_POSSIBLY_COMPLETE_SILENCE_LENGTH_MILLIS, 1000)
    intent.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS, 1500)
    intent.putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, this.getPackageName())
    
    local callbacks = RecognitionListener{
        onReadyForSpeech = function() end,
        onResults = function(data)
            if silenceTimer then mainHandler.removeCallbacks(silenceTimer); silenceTimer = nil end
            local capturedIsDynamic = isDynamicTranslate
            local capturedSrcLang = srcLang
            local capturedTgtLang = tgtLang
            local capturedActualLang = actualLang
            
            local capturedEditText = service.getEditText()
            if not capturedEditText then
                speak("Text box lost. Please tap on text box again.")
                isListening = false
                pcall(function()
                    if currentRecognizer == recognizer then currentRecognizer = nil end
                    recognizer.destroy()
                end)
                return
            end
            
            local matches = nil
            pcall(function() matches = data.getParcelableArrayList(SpeechRecognizer.RESULTS_RECOGNITION) end)
            if not matches then
                pcall(function() matches = data.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION) end)
            end
            
            local recognizedText = nil
            if matches and matches.size() > 0 then recognizedText = matches.get(0) end
            if (not recognizedText or recognizedText == "") and lastPartialText and lastPartialText ~= "" then
                recognizedText = lastPartialText
            end
            if not recognizedText or recognizedText == "" then
                speak("No speech detected. Please try again.")
                isListening = false
                pcall(function()
                    if currentRecognizer == recognizer then currentRecognizer = nil end
                    recognizer.destroy()
                end)
                return
            end
            
            isListening = false
            isTranslating = true
            
            local finalText = recognizedText
            local finalSrc = capturedIsDynamic and capturedSrcLang or capturedActualLang
            local finalTgt = capturedIsDynamic and capturedTgtLang or capturedActualLang
            local finalMode = capturedIsDynamic and "translate" or "typing"
            
            schedulePendingTranslation(finalText, finalSrc, finalTgt, finalMode, capturedEditText)
            
            mainHandler.postDelayed(function()
                pcall(function()
                    if currentRecognizer == recognizer then currentRecognizer = nil end
                    recognizer.destroy()
                end)
                isTranslating = false
            end, 3000)
        end,
        onPartialResults = function(data)
            local partials = nil
            pcall(function() partials = data.getParcelableArrayList(SpeechRecognizer.RESULTS_RECOGNITION) end)
            if not partials then
                pcall(function() partials = data.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION) end)
            end
            if partials and partials.size() > 0 then
                local pt = partials.get(0)
                if pt and pt ~= "" then 
                    lastPartialText = pt
                    resetSilenceTimer(recognizer) 
                end
            end
        end,
        onError = function(error)
            if silenceTimer then mainHandler.removeCallbacks(silenceTimer); silenceTimer = nil end
            
            if error == 5 or error == SpeechRecognizer.ERROR_CLIENT then
                if isTranslating then
                    return
                end
            end
            
            if lastPartialText and lastPartialText ~= "" and isListening then
                local capturedEditText = service.getEditText()
                if capturedEditText then
                    local capturedIsDynamic = isDynamicTranslate
                    local capturedSrcLang = srcLang
                    local capturedTgtLang = tgtLang
                    local capturedActualLang = actualLang
                    local finalSrc = capturedIsDynamic and capturedSrcLang or capturedActualLang
                    local finalTgt = capturedIsDynamic and capturedTgtLang or capturedActualLang
                    local finalMode = capturedIsDynamic and "translate" or "typing"
                    isTranslating = true
                    schedulePendingTranslation(lastPartialText, finalSrc, finalTgt, finalMode, capturedEditText)
                    mainHandler.postDelayed(function() isTranslating = false end, 3000)
                end
            end
            
            isListening = false
            pcall(function()
                if currentRecognizer == recognizer then currentRecognizer = nil end
                recognizer.destroy()
            end)
            return
        end,
        onEndOfSpeech = function() 
            if isListening and not isTranslating and currentRecognizer == recognizer then
                mainHandler.postDelayed(function()
                    if isListening and not isTranslating and currentRecognizer == recognizer then
                        pcall(function()
                            recognizer.startListening(intent)
                        end)
                    end
                end, 1000)
            end
        end,
        onBufferReceived = function(buffer) 
            if isListening and not isTranslating then resetSilenceTimer(recognizer) end 
        end
    }
    
    recognizer.setRecognitionListener(callbacks)
    recognizer.startListening(intent)
end

function stopVoice()
  cleanupOldRecognizer()
end

-- 📖 WORD LIST (Search + Add + Delete)
-- ============================================
function showWordList(lang)
    local langDisplayName = "Language"
    for _, l in ipairs(translationLangs) do
        if l.code == lang then langDisplayName = l.name end
    end
    speak("Showing " .. langDisplayName .. " words")
    
    local word_dlg = LuaDialog(this).setTitle("Words — " .. langDisplayName)
    local w_layout = LinearLayout(this).setOrientation(1).setPadding(40,40,40,40)
    
    local searchBox = EditText(this)
    searchBox.setHint("Search word...")
    searchBox.setSingleLine(true)
    w_layout.addView(searchBox)
    
    local countText = TextView(this)
    countText.setText("Total: " .. getWordCount(lang) .. " words")
    countText.setTextSize(13)
    countText.setTextColor(0xFF666666)
    countText.setPadding(0,10,0,10)
    w_layout.addView(countText)
    
    local listContainer = LinearLayout(this).setOrientation(1)
    w_layout.addView(listContainer)
    
    local function refreshList()
        listContainer.removeAllViews()
        
        local searchTerm = ""
        pcall(function() searchTerm = searchBox.getText().toString():lower() end)
        
        local wordItems = {}
        for word, meaning in pairs(wordBanks[lang] or {}) do
            if searchTerm == "" or word:lower():find(searchTerm, 1, true) or meaning:lower():find(searchTerm, 1, true) then
                table.insert(wordItems, {w = word, m = meaning})
            end
        end
        
        table.sort(wordItems, function(a, b) return a.w < b.w end)
        
        if #wordItems == 0 then
            local empty = TextView(this)
            empty.setText("No words found")
            empty.setTextSize(14)
            empty.setTextColor(0xFF999999)
            empty.setGravity(Gravity.CENTER)
            empty.setPadding(0,40,0,40)
            listContainer.addView(empty)
            return
        end
        
        for _, item in ipairs(wordItems) do
            local row = LinearLayout(this).setOrientation(0)
            row.setPadding(0,10,0,10)
            
            local wordText = TextView(this)
            wordText.setText(item.w .. " → " .. item.m)
            wordText.setTextSize(15)
            wordText.setLayoutParams(LinearLayout.LayoutParams(0, -2, 1))
            row.addView(wordText)
            
            local delBtn = Button(this)
            delBtn.setText("❌")
            delBtn.setTextSize(12)
            delBtn.setOnClickListener(function()
                wordBanks[lang][item.w] = nil
                saveDictionary(lang)
                speak("Deleted")
                refreshList()
                countText.setText("Total: " .. getWordCount(lang) .. " words")
            end)
            row.addView(delBtn)
            
            listContainer.addView(row)
        end
    end
    
    searchBox.addTextChangedListener{
        onTextChanged = function() refreshList() end
    }
    
    w_layout.addView(TextView(this).setText("Add New Word:").setTextSize(15).setTextColor(0xFF000000).setPadding(0,20,0,5))
    
    local newWordBox = EditText(this)
    newWordBox.setHint("New word")
    newWordBox.setSingleLine(true)
    w_layout.addView(newWordBox)
    
    local newMeaningBox = EditText(this)
    newMeaningBox.setHint("Meaning")
    newMeaningBox.setSingleLine(true)
    w_layout.addView(newMeaningBox)
    
    local addBtn = Button(this)
    addBtn.setText("➕ Add Word")
    addBtn.setOnClickListener(function()
        local newWord = ""
        local newMeaning = ""
        pcall(function()
            newWord = newWordBox.getText().toString():gsub("^%s+", ""):gsub("%s+$", "")
            newMeaning = newMeaningBox.getText().toString():gsub("^%s+", ""):gsub("%s+$", "")
        end)
        
        if newWord == "" or newMeaning == "" then
            speak("Please type word and meaning")
            return
        end
        
        if not wordBanks[lang] then wordBanks[lang] = {} end
        wordBanks[lang][newWord] = newMeaning
        saveDictionary(lang)
        speak("Word added")
        
        pcall(function()
            newWordBox.setText("")
            newMeaningBox.setText("")
        end)
        
        refreshList()
        countText.setText("Total: " .. getWordCount(lang) .. " words")
    end)
    w_layout.addView(addBtn)
    
    refreshList()
    
    local scroll = ScrollView(this)
    scroll.addView(w_layout)
    
    word_dlg.setView(scroll)
    word_dlg.setPositiveButton("Close", function() speak("Closed") end)
    word_dlg.show()
end

    if position == 0 then 
        selectedLang = "en-US"

        config.targetLang = "en"
    elseif position == 1 then 
        selectedLang = "ur-PK"
        config.sourceLang = "ur"
        config.targetLang = "ur"
    elseif position == 2 then 
        selectedLang = "hi-IN"

        config.targetLang = "hi"
    elseif position == 3 then 
        selectedLang = "pa-IN"
        config.sourceLang = "pa"
        config.targetLang = "pa"
    elseif position == 4 then 
        selectedLang = "ar-SA"
        config.sourceLang = "ar"
        config.targetLang = "ar"
    elseif position == 5 then

        isTranslationMode = true

        config.sourceLang = pref.getString("cfg_sourceLang", "ur")
        config.targetLang = pref.getString("cfg_targetLang", "en")
    end

    if not isTranslationMode then
        pref.edit().putString("cfg_sourceLang", config.sourceLang).apply()
        pref.edit().putString("cfg_targetLang", config.targetLang).apply()
    end
    
    local selectedName = buildLangList()[position + 1]
    
    -- ✅ Translation پر مکمل نام بولیں
    if isTranslationMode then
        local srcName = getLangName(config.sourceLang)
        local tgtName = getLangName(config.targetLang)
        speak("Translation " .. srcName .. " to " .. tgtName)
    else
        speak("Selected " .. selectedName)
    end
    
    main_dlg.dismiss()
    mainHandler.postDelayed(function()

    end, 300)
end
